/*******************************************************************
 * 자산 태그 매니저 <-> 구글 시트 양방향 동기화 (Google Apps Script)
 *
 * 설치:
 *  1) 구글 시트 메뉴 "확장 프로그램 > Apps Script"에 이 코드를 전부 붙여 넣고 저장
 *  2) 아래 SERVICE_KEY 에 Supabase "service_role" (secret) 키를 붙여 넣기
 *     - Supabase > Project Settings > API Keys > "service_role" 또는 "Secret keys"
 *     - 이 키는 이 스크립트 안에만 두고 절대 외부에 공유하지 말 것
 *  3) 위쪽 함수 선택 상자에서 setup 을 고르고 ▶ 실행 → 권한 허용
 *  4) 시트에 "자산" 탭이 생기고 1분마다 자동 동기화가 시작됨
 *
 * 동작:
 *  - 시트에서 셀을 고치면 즉시 Supabase에 반영 (대시보드는 30초 안에 받아감)
 *  - 대시보드에서 바꾸면 1분 안에 시트에 반영
 *  - 토큰 칸이 비어 있는 새 행은 자동으로 토큰이 발급되고, 자산번호가 비어 있으면 자동 채번됨
 *  - "삭제" 칸에 체크하면 대시보드에서도 삭제됨 (행을 직접 지우는 건 반영되지 않음)
 *******************************************************************/

var SUPABASE_URL = "https://mczvdfcwnsfxohprqfgq.supabase.co";
var SERVICE_KEY  = "여기에_service_role_키_붙여넣기";
var SHEET_NAME   = "자산";

var HEADERS = ["토큰","장비","분류","자산번호","시리얼","담당자","소속팀","지급일","상태","취득가(원)","비고","삭제","수정시각","출처"];
var COL = {}; HEADERS.forEach(function(h, i){ COL[h] = i; });
var CATEGORIES = ["노트북","모니터","마우스","키보드","태블릿","헤드셋","도킹/허브","기타"];
var STATUSES   = ["사용중","미사용","수리중","회수됨"];

/* ---------- 설치 ---------- */
function setup(){
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName(SHEET_NAME) || ss.insertSheet(SHEET_NAME);
  if(sh.getLastRow() === 0 || sh.getRange(1,1).getValue() !== HEADERS[0]){
    sh.clear();
    sh.getRange(1,1,1,HEADERS.length).setValues([HEADERS]).setFontWeight("bold").setBackground("#eaeef2");
    sh.setFrozenRows(1);
  }
  // 입력 도움: 분류/상태 드롭다운, 삭제 체크박스
  var maxRows = Math.max(sh.getMaxRows(), 1000);
  sh.getRange(2, COL["분류"]+1, maxRows-1, 1).setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(CATEGORIES, true).setAllowInvalid(true).build());
  sh.getRange(2, COL["상태"]+1, maxRows-1, 1).setDataValidation(SpreadsheetApp.newDataValidation().requireValueInList(STATUSES, true).setAllowInvalid(true).build());
  sh.getRange(2, COL["삭제"]+1, maxRows-1, 1).insertCheckboxes();
  sh.getRange(2, COL["토큰"]+1, maxRows-1, 1).setFontFamily("Courier New").setFontColor("#777777");
  sh.getRange(2, COL["수정시각"]+1, maxRows-1, 2).setFontColor("#999999");
  sh.getRange(2, COL["지급일"]+1, maxRows-1, 1).setNumberFormat("@");
  sh.getRange(2, COL["취득가(원)"]+1, maxRows-1, 1).setNumberFormat("#,##0");
  sh.autoResizeColumns(1, HEADERS.length);

  // 트리거 재설치
  ScriptApp.getProjectTriggers().forEach(function(t){ ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger("onEditHandler").forSpreadsheet(ss).onEdit().create();
  ScriptApp.newTrigger("pullFromSupabase").timeBased().everyMinutes(1).create();

  pullFromSupabase();
  SpreadsheetApp.getUi().alert("설정 완료. '" + SHEET_NAME + "' 탭에서 자산을 관리하세요. 1분마다 자동 동기화됩니다.");
}

/* ---------- Supabase 호출 ---------- */
function sb(path, method, payload, extraHeaders){
  var headers = { "apikey": SERVICE_KEY, "Authorization": "Bearer " + SERVICE_KEY, "Content-Type": "application/json" };
  if(extraHeaders) for(var k in extraHeaders) headers[k] = extraHeaders[k];
  var res = UrlFetchApp.fetch(SUPABASE_URL + "/rest/v1/" + path, { method: method || "get", headers: headers, payload: payload ? JSON.stringify(payload) : null, muteHttpExceptions: true });
  var code = res.getResponseCode();
  if(code >= 300) throw new Error("Supabase " + code + ": " + res.getContentText().slice(0, 300));
  var txt = res.getContentText();
  return txt ? JSON.parse(txt) : null;
}

/* ---------- 자산번호 자동 발급 (Supabase 채번기) ---------- */
var PREFIX = { "노트북":"NB", "모니터":"MN", "마우스":"MS", "키보드":"KB", "태블릿":"TB", "헤드셋":"HS", "도킹/허브":"DK", "기타":"ET" };
function nextAssetId(category){
  var r = sb("rpc/next_asset_id", "post", { prefix: PREFIX[category] || "ET" });
  return typeof r === "string" ? r : String(r);
}

/* ---------- 유틸 ---------- */
function genToken(){
  var chars = "abcdefghjkmnpqrstuvwxyz23456789", out = "";
  for(var i=0;i<12;i++) out += chars[Math.floor(Math.random()*chars.length)];
  return out.match(/.{1,4}/g).join("-");
}
function fmtDate(v){
  if(!v) return "";
  if(Object.prototype.toString.call(v) === "[object Date]"){ return Utilities.formatDate(v, Session.getScriptTimeZone(), "yyyy-MM-dd"); }
  return String(v).trim().replace(/[./]/g, "-").slice(0, 10);
}
function rowToRecord(row){
  return {
    token: String(row[COL["토큰"]] || "").trim(),
    name: String(row[COL["장비"]] || "").trim(),
    category: String(row[COL["분류"]] || "").trim() || "기타",
    asset_id: String(row[COL["자산번호"]] || "").trim(),
    serial: String(row[COL["시리얼"]] || "").trim(),
    holder_name: String(row[COL["담당자"]] || "").trim(),
    holder_dept: String(row[COL["소속팀"]] || "").trim(),
    issue_date: fmtDate(row[COL["지급일"]]),
    status: String(row[COL["상태"]] || "").trim() || "사용중",
    value: Math.round(Number(String(row[COL["취득가(원)"]] || "0").replace(/[^0-9.]/g, "")) || 0),
    note: String(row[COL["비고"]] || "").trim(),
    deleted: row[COL["삭제"]] === true,
    source: "sheet",
    updated_at: new Date().toISOString()
  };
}
function recordToRow(r){
  var row = new Array(HEADERS.length).fill("");
  row[COL["토큰"]] = r.token; row[COL["장비"]] = r.name || ""; row[COL["분류"]] = r.category || "";
  row[COL["자산번호"]] = r.asset_id || ""; row[COL["시리얼"]] = r.serial || ""; row[COL["담당자"]] = r.holder_name || "";
  row[COL["소속팀"]] = r.holder_dept || ""; row[COL["지급일"]] = r.issue_date || ""; row[COL["상태"]] = r.status || "";
  row[COL["취득가(원)"]] = Number(r.value) || 0; row[COL["비고"]] = r.note || ""; row[COL["삭제"]] = false;
  row[COL["수정시각"]] = r.updated_at ? Utilities.formatDate(new Date(r.updated_at), Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm") : "";
  row[COL["출처"]] = r.source === "sheet" ? "시트" : "대시보드";
  return row;
}

/* ---------- 시트 -> Supabase (편집 즉시) ---------- */
function onEditHandler(e){
  try {
    var sh = e.range.getSheet();
    if(sh.getName() !== SHEET_NAME) return;
    var r1 = e.range.getRow(), r2 = r1 + e.range.getNumRows() - 1;
    if(r2 < 2) return;
    r1 = Math.max(r1, 2);
    var lock = LockService.getScriptLock(); lock.waitLock(10000);
    try {
      var recs = [];
      for(var r = r1; r <= r2; r++){
        var row = sh.getRange(r, 1, 1, HEADERS.length).getValues()[0];
        var rec = rowToRecord(row);
        if(!rec.name && !rec.token) continue;               // 완전히 빈 행
        if(!rec.token){ rec.token = genToken(); sh.getRange(r, COL["토큰"]+1).setValue(rec.token); }
        if(!rec.name) continue;                              // 이름 없으면 보류
        if(!rec.asset_id){ rec.asset_id = nextAssetId(rec.category); sh.getRange(r, COL["자산번호"]+1).setValue(rec.asset_id); }
        recs.push(rec);
        sh.getRange(r, COL["수정시각"]+1, 1, 2).setValues([[Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm"), "시트"]]);
      }
      if(recs.length){
        sb("assets?on_conflict=token", "post", recs, { "Prefer": "resolution=merge-duplicates,return=minimal" });
        // 삭제 체크된 행은 시트에서도 제거 (대시보드가 30초 안에 삭제를 받아감)
        for(var r = r2; r >= r1; r--){ if(sh.getRange(r, COL["삭제"]+1).getValue() === true) sh.deleteRow(r); }
      }
    } finally { lock.releaseLock(); }
  } catch(err){
    SpreadsheetApp.getActiveSpreadsheet().toast("동기화 실패: " + err.message, "자산 동기화", 8);
  }
}

/* ---------- Supabase -> 시트 (1분마다) ---------- */
function pullFromSupabase(){
  var lock = LockService.getScriptLock(); if(!lock.tryLock(20000)) return;
  try {
    var sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME); if(!sh) return;
    var rows = sb("assets?select=*&deleted=eq.false&order=updated_at.desc", "get") || [];
    var byToken = {}; rows.forEach(function(r){ byToken[r.token] = r; });

    var last = sh.getLastRow();
    var existing = last >= 2 ? sh.getRange(2, 1, last-1, HEADERS.length).getValues() : [];
    var seen = {}, toDelete = [];
    for(var i=0;i<existing.length;i++){
      var tok = String(existing[i][COL["토큰"]] || "").trim();
      if(!tok) continue;                       // 아직 저장 안 된 새 행은 건드리지 않음
      var r = byToken[tok];
      if(!r){ toDelete.push(i+2); continue; }  // Supabase에서 삭제됨
      seen[tok] = true;
      var newRow = recordToRow(r);
      var cur = existing[i];
      var changed = false;
      for(var c=0;c<HEADERS.length;c++){ if(c === COL["삭제"]) continue; if(String(cur[c]) !== String(newRow[c])){ changed = true; break; } }
      if(changed){ newRow[COL["삭제"]] = cur[COL["삭제"]] === true; sh.getRange(i+2, 1, 1, HEADERS.length).setValues([newRow]); }
    }
    var toAppend = rows.filter(function(r){ return !seen[r.token]; }).map(recordToRow);
    if(toAppend.length){ sh.getRange(sh.getLastRow()+1, 1, toAppend.length, HEADERS.length).setValues(toAppend); sh.getRange(sh.getLastRow()-toAppend.length+1, COL["삭제"]+1, toAppend.length, 1).insertCheckboxes(); }
    toDelete.sort(function(a,b){ return b-a; }).forEach(function(rn){ sh.deleteRow(rn); });
  } finally { lock.releaseLock(); }
}

/* ---------- 수동 메뉴 ---------- */
function onOpen(){
  SpreadsheetApp.getUi().createMenu("자산 동기화")
    .addItem("지금 가져오기 (대시보드 → 시트)", "pullFromSupabase")
    .addItem("전체 올리기 (시트 → 대시보드)", "pushAllToSupabase")
    .addToUi();
}
function pushAllToSupabase(){
  var sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME); if(!sh) return;
  var last = sh.getLastRow(); if(last < 2) return;
  var vals = sh.getRange(2, 1, last-1, HEADERS.length).getValues(), recs = [];
  for(var i=0;i<vals.length;i++){
    var rec = rowToRecord(vals[i]); if(!rec.name) continue;
    if(!rec.token){ rec.token = genToken(); sh.getRange(i+2, COL["토큰"]+1).setValue(rec.token); }
    if(!rec.asset_id){ rec.asset_id = nextAssetId(rec.category); sh.getRange(i+2, COL["자산번호"]+1).setValue(rec.asset_id); }
    recs.push(rec);
  }
  for(var k=0;k<recs.length;k+=200) sb("assets?on_conflict=token", "post", recs.slice(k, k+200), { "Prefer": "resolution=merge-duplicates,return=minimal" });
  SpreadsheetApp.getActiveSpreadsheet().toast(recs.length + "건을 올렸습니다.", "자산 동기화", 5);
}
